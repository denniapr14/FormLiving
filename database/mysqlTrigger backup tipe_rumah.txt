    DECLARE max_tidur INT;
    DECLARE max_mandi INT;

    DECLARE var_max_tidur INT;
    DECLARE var_max_mandi INT;

    SELECT max_harga_rumah, max_kamar_tidur, max_kamar_mandi
    INTO var_max_harga, var_max_tidur, var_max_mandi
    FROM var_tipe_rumah
    WHERE id_var_tipe_rumah = 1;

    IF NEW.harga_tr > var_max_harga THEN
        SET max_harga = NEW.harga_tr;

        -- Update the var_tipe_rumah table with the new max_harga
        UPDATE var_tipe_rumah
        SET max_harga_rumah = max_harga
        WHERE id_var_tipe_rumah = 1;
    END IF;

    IF NEW.kmr_tidur_tr > var_max_tidur THEN

        UPDATE var_tipe_rumah
        SET max_kamar_tidur = max_tidur
        WHERE id_var_tipe_rumah = 1;

        SET max_mandi = NEW.kmr_mandi_tr;

        -- Update the var_tipe_rumah table with the new max_kamar_mandi
        UPDATE var_tipe_rumah
        SET max_kamar_mandi = max_mandi
        WHERE id_var_tipe_rumah = 1;
    END IF;

    -- Check and update max_harga if NEW.harga_tr is greater than or equal to var_max_harga
    IF old.harga_tr >= var_max_harga THEN

        WHERE id_var_tipe_rumah = 1;
    END IF;

    -- Check and update max_tidur if NEW.kmr_tidur_tr is greater than or equal to var_max_tidur
    IF old.kmr_tidur_tr >= var_max_tidur THEN
        SET max_tidur = var_max_tidur;

        UPDATE var_tipe_rumah
        SET max_kamar_tidur = max_tidur
        WHERE id_var_tipe_rumah = 1;

        -- Update the var_tipe_rumah table with the new max_kamar_mandi

        -- Update the var_tipe_rumah table with the new max_harga
        UPDATE var_tipe_rumah
        SET max_harga_rumah = max_harga
        WHERE id_var_tipe_rumah = 1;

        SET max_kamar_tidur = max_tidur
        WHERE id_var_tipe_rumah = 1;
